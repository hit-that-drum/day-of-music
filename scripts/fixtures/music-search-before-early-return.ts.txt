// /api/music/search — proxy for the free iTunes Search API.
//
// No auth or API key required. Results are mapped to a compact shape the
// client can turn into an Album. Identical queries are cached for an hour to
// stay well under the iTunes rate limit (~20 calls/min).
//
// The iTunes catalog IDs are Apple Music catalog IDs, so this can later be
// swapped for the (paid) Apple Music API without changing the client.

import dayjs from "dayjs";
import { z } from "zod";

const searchParamsSchema = z.object({
  q: z.string().trim().min(1).max(300),
  type: z.enum(["album", "artist", "track"]).default("album"),
  limit: z.coerce.number().int().min(1).max(200).default(12),
  country: z
    .string()
    .regex(/^[a-zA-Z]{2}$/, "country must be a 2-letter code")
    .default("US"),
});

const ENTITY_BY_TYPE = {
  album: "song,album",
  artist: "musicArtist",
  track: "song",
} as const;

type ITunesResult = {
  wrapperType?: string;
  kind?: string;
  collectionCensoredName?: string;
  trackCensoredName?: string;
  collectionId?: number;
  trackId?: number;
  artistId?: number;
  collectionName?: string;
  trackName?: string;
  artistName?: string;
  primaryGenreName?: string;
  releaseDate?: string;
  artworkUrl100?: string;
  trackCount?: number;
};

export type MusicSearchResult = {
  id: string;
  title: string;
  artist: string;
  genre: string;
  year: number;
  releaseDate: string;
  artworkUrl: string;
  trackCount: number;
};

function preferredTitle(name = "", alternate = ""): string {
  // Preserve version labels such as (Live) and [Remastered]. Observed in
  // Odesli's public client; this chooses metadata, not search relevance.
  const annotations = /[[(][^)\]]+[\])]/g;
  return alternate.length > name.length &&
    (alternate.match(annotations)?.length ?? 0) > (name.match(annotations)?.length ?? 0)
    ? alternate : name;
}

function mapResult(r: ITunesResult, type: "album" | "artist" | "track"): MusicSearchResult | null {
  const rawId = type === "track" ? r.trackId : type === "artist" ? r.artistId : r.collectionId;
  const title = type === "track" ? preferredTitle(r.trackName, r.trackCensoredName)
    : type === "artist" ? r.artistName
    : preferredTitle(r.collectionName, r.collectionCensoredName);
  if (rawId == null || !title) return null;

  const releaseDate = r.releaseDate ?? "";
  return {
    id: `itunes-${rawId}`,
    title,
    artist: r.artistName ?? "",
    genre: r.primaryGenreName ?? "—",
    year: releaseDate ? dayjs(releaseDate).year() : 0,
    releaseDate,
    // Artwork URLs are templated; request a sharper 600x600 variant.
    artworkUrl: r.artworkUrl100?.replace("100x100bb", "600x600bb") ?? "",
    trackCount: r.trackCount ?? 0,
  };
}

export async function GET(request: Request) {
  const url = new URL(request.url);
  const parsed = searchParamsSchema.safeParse(
    Object.fromEntries(url.searchParams),
  );

  if (!parsed.success) {
    return Response.json(
      {
        error: "Invalid search parameters.",
        issues: parsed.error.flatten().fieldErrors,
      },
      { status: 400 },
    );
  }

  const { q, type, limit, country } = parsed.data;

  // Two bounded stages: preferred store, then regional alternatives in parallel.
  // A single mixed-entity request also finds albums by their song titles.
  const preferred = country.toUpperCase();
  const storefronts = [...new Set([preferred, "US", "KR", "JP", "GB"])];

  async function searchIn(store: string): Promise<MusicSearchResult[] | null> {
    const itunesUrl = new URL("https://itunes.apple.com/search");
    itunesUrl.searchParams.set("term", q);
    itunesUrl.searchParams.set("media", "music");
    itunesUrl.searchParams.set("entity", ENTITY_BY_TYPE[type]);
    itunesUrl.searchParams.set("limit", String(type === "album" ? Math.min(200, Math.max(50, limit * 3)) : limit));
    itunesUrl.searchParams.set("country", store);
    try {
      const resp = await fetch(itunesUrl, {
        next: { revalidate: 3600 },
        signal: AbortSignal.any([request.signal, AbortSignal.timeout(4000)]),
      });
      if (!resp.ok) return null;
      const body = await resp.json() as { results?: ITunesResult[] };
      if (!Array.isArray(body.results)) return null;
      const rows = body.results.filter((r) => type === "album"
        ? r.wrapperType === "collection" || (r.wrapperType === "track" && r.kind === "song")
        : type === "track" ? r.kind === "song" : r.wrapperType === "artist");
      const localized = new Map<number, ITunesResult>();
      for (const row of rows) {
        if (row.kind === "song" && row.collectionId != null && row.collectionName) {
          if (!localized.has(row.collectionId)) localized.set(row.collectionId, row);
        }
      }
      const seen = new Set<string>();
      return rows.map((row) => {
        const song = row.collectionId == null ? undefined : localized.get(row.collectionId);
        return mapResult(type === "album" && song ? {
          ...row, collectionName: song.collectionName,
          collectionCensoredName: song.collectionCensoredName,
        } : row, type);
      }).filter((row): row is MusicSearchResult => {
        if (!row || seen.has(row.id)) return false;
        seen.add(row.id);
        return true;
      }).slice(0, limit);
    } catch {
      return null;
    }
  }

  const first = await searchIn(preferred);
  if (first?.length) return Response.json({ results: first });
  if (request.signal.aborted) return Response.json({ error: "Search cancelled." }, { status: 408 });
  const fallback = await Promise.all(storefronts.slice(1).map(searchIn));
  const seen = new Set<string>();
  const results = fallback.flatMap((rows) => rows ?? []).filter((row) => {
    if (seen.has(row.id)) return false;
    seen.add(row.id);
    return true;
  }).slice(0, limit);
  if (first !== null || fallback.some((rows) => rows !== null)) {
    return Response.json({ results });
  }
  return Response.json({ error: "Music search failed." }, { status: 502 });
}
